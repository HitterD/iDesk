# ==============================================================================
# Snippet untuk Tab "Advanced" di Nginx Proxy Manager (NPM)
# Domain: idesk.santos.co.id -> Forward Host: 192.168.2.136, Port: 3080

# 1. Izinkan upload file besar (lampiran tiket, dokumen, installer)
client_max_body_size 100M;

# 2. Timeout untuk laporan data besar / proses background
proxy_connect_timeout 300s;
proxy_send_timeout 300s;
proxy_read_timeout 300s;

# 3. Buffer tuning untuk respon JSON besar
proxy_buffering on;
proxy_buffer_size 128k;
proxy_buffers 4 256k;
proxy_busy_buffers_size 256k;

# 4. Header tambahan untuk memastikan Client IP asli terbaca iDesk
proxy_set_header X-Forwarded-Host $host;
proxy_set_header X-Forwarded-Port $server_port;
